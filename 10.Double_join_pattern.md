# Answer
```sql

-- Select each unique member and the facility they booked
SELECT DISTINCT
    CONCAT(membs.firstname, ' ', membs.surname) AS member,
    fcs.name AS facility

-- Start with the members table
FROM cd.members membs

-- Join members to their bookings
JOIN cd.bookings bkgs
    ON membs.memid = bkgs.memid

-- Join bookings to the facilities they booked
JOIN cd.facilities fcs
    ON bkgs.facid = fcs.facid

-- Only include the two tennis courts
WHERE fcs.name IN (
    'Tennis Court 1',
    'Tennis Court 2'
)

-- Sort alphabetically by member, then facility
ORDER BY
    member,
    facility;
```
