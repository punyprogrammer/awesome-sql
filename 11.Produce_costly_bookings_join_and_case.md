# Solution
```sql
-- Select the member's full name, facility name, and total booking cost
SELECT
    members.firstname || ' ' || members.surname AS member,
    facilities.name AS facility,

    -- Guests (memid = 0) use guestcost;
    -- regular members use membercost
    CASE
        WHEN members.memid = 0
            THEN facilities.guestcost * bookings.slots
        ELSE
            facilities.membercost * bookings.slots
    END AS cost

-- Start with the members table
FROM cd.members members

-- Join bookings to identify which member made each booking
JOIN cd.bookings bookings
    ON members.memid = bookings.memid

-- Join facilities to get the facility name and pricing
JOIN cd.facilities facilities
    ON bookings.facid = facilities.facid

WHERE
    -- Only include bookings made on September 14, 2012
    bookings.starttime >= '2012-09-14'
    AND bookings.starttime < '2012-09-15'

    -- Booking must cost more than $30
    AND (
        -- Guest: use guest cost per slot
        (
            members.memid = 0
            AND facilities.guestcost * bookings.slots > 30
        )

        OR

        -- Member: use member cost per slot
        (
            members.memid > 0
            AND facilities.membercost * bookings.slots > 30
        )
    )

-- Show the most expensive bookings first
ORDER BY cost DESC;
```
